% Frise : noms courts des sections précédentes, points de la section courante (nom court centré dessous), noms des suivantes
\def\deckgendots{\begingroup\tiny
  \def\subsectionentry##1##2##3##4##5{}%
  % passe 1 : noms des autres sections, et le point « en cours » = dernier transparent compté ne dépassant pas celui-ci
  \deckgen@avant={}\deckgen@apres={}\deckgen@courant=0\relax
  \def\sectionentry##1##2##3##4##5{\ifnum##1<\c@section\relax\deckgen@avant=\expandafter{\the\deckgen@avant{\color{gris}##2}\hskip8pt plus1fil}\fi
    \ifnum##1>\c@section\relax\deckgen@apres=\expandafter{\the\deckgen@apres\hskip8pt plus1fil{\color{gris}##2}}\fi}%
  \def\slideentry##1##2##3##4##5##6{\ifnum\c@section=##1\relax\ifnum##3>\c@subsectionslide\else\deckgen@courant=##3\relax\fi\fi}%
  \dohead
  % passe 2 : les points
  \def\sectionentry##1##2##3##4##5{}%
  \def\slideentry##1##2##3##4##5##6{\ifnum\c@section=##1\relax
    \ifnum\deckgen@courant=##3\relax{\color{teinte}\usebeamertemplate{mini frame}}%
    \else{\color{bleugris!60}\usebeamertemplate{mini frame in current subsection}}\fi\hskip\beamer@boxsize minus0.8\beamer@boxsize\fi}%
  \setbox\deckgen@dotsbox=\hbox{\dohead\unskip}%
  % si les points et les noms ne tiennent pas sur la ligne, on resserre les points (jusqu'à 80 % de leur écart)
  \setbox\deckgen@avantbox=\hbox{\the\deckgen@avant}\setbox\deckgen@apresbox=\hbox{\the\deckgen@apres}%
  \deckgen@dispo=\dimexpr\hsize-\leftskip-\rightskip-\wd\deckgen@avantbox-\wd\deckgen@apresbox\relax
  \ifdim\wd\deckgen@dotsbox>\deckgen@dispo\setbox\deckgen@dotsbox=\hbox to\deckgen@dispo{\dohead\unskip}\fi
  \hbox to\dimexpr\hsize-\leftskip-\rightskip\relax{\the\deckgen@avant
    \vtop{\hbox{\copy\deckgen@dotsbox}\vskip1pt\hbox to\wd\deckgen@dotsbox{\hss{\color{teinte}(\ifnum\c@section=0 \color{gris}hors section\else\insertsectionhead\fi)}\hss}}%
    \the\deckgen@apres}%
  \endgroup}
